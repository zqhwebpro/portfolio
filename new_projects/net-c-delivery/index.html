/**
 * Delivery Gateway Configuration for WordPress
 * Generated based on active API credentials
 */
define('DELIVERY_GATEWAY_ACTIVE', true);
define('DELIVERY_GATEWAY_PROVIDER', 'TOAST_POS');
define('DELIVERY_GATEWAY_CLIENT_ID', 'toast_app_dinedispatch_982');
define('DELIVERY_GATEWAY_LOCATION_ID', 'loc_downtown_toast_01');
define('DELIVERY_GATEWAY_TLS_VERIFIED', true);

// WordPress Action Hook: Automatically dispatch orders to connected delivery fleet
add_action('woocommerce_checkout_order_processed', function($order_id) {
    $order = wc_get_order($order_id);
    Delivery_Gateway_Client::dispatch_ticket($order);
});
