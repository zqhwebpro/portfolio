Custom Session Cart Model Binder

public class CartModelBinder : IModelBinder
{
    public object BindModel(ControllerContext cc, ModelBindingContext bc) {
        Cart cart = (Cart)cc.HttpContext.Session["Cart"] ?? new Cart();
        cc.HttpContext.Session["Cart"] = cart;
        return cart;
    }
}
